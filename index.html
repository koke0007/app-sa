// ── Fecha normalizer ──────────────────────────────────────────

function normalizeFecha(raw){

  if(!raw) return '';

  const s=String(raw).trim().replace(/[-]/g,'/');

  // aaaa/mm/dd

  if(/^\d{4}\/\d{1,2}\/\d{1,2}$/.test(s)){

    const[y,m,d]=s.split('/');

    return `${y}-${m.padStart(2,'0')}-${d.padStart(2,'0')}`;

  }

  // dd/mm/aaaa

  if(/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(s)){

    const[d,m,y]=s.split('/');

    return `${y}-${m.padStart(2,'0')}-${d.padStart(2,'0')}`;

  }

  // dd/mm/aa

  if(/^\d{1,2}\/\d{1,2}\/\d{2}$/.test(s)){

    const[d,m,y]=s.split('/');

    const full=parseInt(y)>30?'19'+y:'20'+y;

    return `${full}-${m.padStart(2,'0')}-${d.padStart(2,'0')}`;

  }

  // ya en formato yyyy-mm-dd

  if(/^\d{4}-\d{2}-\d{2}$/.test(String(raw))) return String(raw);

  return '';

}


function getEdad(raw){

  const fnac=normalizeFecha(raw);

  if(!fnac) return '—';

  const h=new Date(),n=new Date(fnac);

  let e=h.getFullYear()-n.getFullYear();

  if(h.getMonth()-n.getMonth()<0||(h.getMonth()===n.getMonth()&&h.getDate()<n.getDate()))e--;

  return e>=0?e:'—';

}


function calcEdad(){

  const v=document.getElementById('p_fnac').value;

  document.getElementById('p_edad').value=v?getEdad(v):'';

}

function getField(p,...keys){

  for(const k of keys){

    const found=Object.keys(p).find(pk=>pk.trim().toLowerCase()===k.trim().toLowerCase());

async function apiCall(action,sheet,row,index){

  let url=`${API}?action=${action}&sheet=${encodeURIComponent(sheet)}`;

}
